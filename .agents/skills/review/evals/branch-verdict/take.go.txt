package pump

import "context"

// Take receives at most n values from p and returns them in arrival order.
// It returns fewer when p's data channel closes or ctx ends first, and never
// blocks past ctx. It does not stop or cancel p.
func Take[T any](ctx context.Context, p *Pump[T], n int) []T {
	var out []T
	for len(out) <= n {
		select {
		case <-ctx.Done():
			return out
		case value, ok := <-p.Data():
			if !ok {
				return out
			}
			out = append(out, value)
		}
	}
	return out
}
